#!/usr/bin/env bash
# AutoFeature Phase 4 -> Kaggle.  Run from the repo root (finds af_phase4.py in ./ or code/).
#
#   ./push_phase4.sh <notebook> [--dry]        (KUSER defaults to yoka33)
#
# notebooks   a_cpu  random, FDR gate, rungs 1+2 x 5 runs                (CPU)  <- headline
#             a_gpu  Qwen-14B (Phase 3 prompt, operator nudge), rungs 1+2 (GPU)  <- headline
#             b_gpu  Qwen-14B WITHOUT the operator nudge, rungs 1+2        (GPU)
#             c_cpu  canary arms (all decoys, proxy bypassed), rung 2      (CPU, 30 tests x 3 runs x 2 gates)
#             seal   ONE-SHOT: opens fresh sealed day 2017-11-07 on every finished stack
# defaults in the script: 24 sign-flip blocks, max 30 full tests per run (wall-clock guard 620 min; resume via KSRC)
#
# optional env:  KSRC="user/af-phase4-a-cpu,user/af-phase4-a-gpu,..."   kernel outputs to attach
#                                                   (seal: all finished notebooks; resume: the notebook itself)
#                P3DATA="user/af-phase3-results"    dataset holding Phase 3 results/final_*.json (rung-0 bridge)
#                WCACHE="user/af-w-dev"             dataset holding W_DEV.npz (skips the ~10 min CSV scan)
set -euo pipefail
KUSER=${KUSER:-yoka33}
NB=${1:?usage: ./push_phase4.sh <a_cpu|a_gpu|b_gpu|c_cpu|seal> [--dry]}
DRY=${2:-}
if [ -z "${SRC:-}" ]; then if [ -f code/af_phase4.py ]; then SRC=code/af_phase4.py; else SRC=af_phase4.py; fi; fi
[ -f "$SRC" ] || { echo "missing $SRC"; exit 1; }

case "$NB" in
  a_cpu) GPU=false; NET=false; RUNGS="1,2"; JOBS="R-FDR:1-5";               N=60; MODE=run  ;;
  a_gpu) GPU=true;  NET=true;  RUNGS="1,2"; JOBS="L-FDR-14B:1-5";           N=60; MODE=run  ;;
  b_gpu) GPU=true;  NET=true;  RUNGS="1,2"; JOBS="LF-FDR-14B:1-5";          N=60; MODE=run  ;;
  c_cpu) GPU=false; NET=false; RUNGS="2";   JOBS="C-naive:1-3,C-FDR:1-3";   N=30; MODE=run  ;;
  seal)  GPU=false; NET=false; RUNGS="1,2"; JOBS="R-FDR:1-5";               N=60; MODE=seal ;;
  *) echo "unknown notebook $NB"; exit 1 ;;
esac

SLUG="af-phase4-${NB//_/-}"
D="kaggle/phase4_${NB}"
mkdir -p "$D"
sed -e "s|^DEFAULT_RUNGS = .*|DEFAULT_RUNGS = \"$RUNGS\"|" \
    -e "s|^DEFAULT_JOBS = .*|DEFAULT_JOBS = \"$JOBS\"|" \
    -e "s|^DEFAULT_N = .*|DEFAULT_N = $N|" \
    -e "s|^DEFAULT_MODE = .*|DEFAULT_MODE = \"$MODE\"   # run \| seal \| report|" \
    "$SRC" > "$D/run.py"
grep -q "^DEFAULT_JOBS = \"$JOBS\"" "$D/run.py" || { echo "sed failed to set DEFAULT_JOBS"; exit 1; }
python3 -m py_compile "$D/run.py"

DS="[]"; KS="[]"
[ -n "${WCACHE:-}" ] && DS="[\"$WCACHE\"]"
if [ "$NB" = "seal" ] && [ -n "${P3DATA:-}" ]; then DS="[\"${WCACHE:-}\",\"$P3DATA\"]"; DS=${DS/\"\",/}; fi
[ -n "${KSRC:-}" ] && KS="[\"${KSRC//,/\",\"}\"]"

cat > "$D/kernel-metadata.json" <<JSON
{
  "id": "$KUSER/$SLUG",
  "title": "$SLUG",
  "code_file": "run.py",
  "language": "python",
  "kernel_type": "script",
  "is_private": "true",
  "enable_gpu": "$GPU",
  "enable_internet": "$NET",
  "dataset_sources": $DS,
  "competition_sources": ["talkingdata-adtracking-fraud-detection"],
  "kernel_sources": $KS
}
JSON
echo "built $D  (rungs=$RUNGS jobs=$JOBS n=$N mode=$MODE gpu=$GPU internet=$NET)"
if [ "$DRY" = "--dry" ]; then echo "dry run: not pushed"; exit 0; fi
kaggle kernels push -p "$D"
echo "status:  kaggle kernels status $KUSER/$SLUG"
echo "output:  kaggle kernels output $KUSER/$SLUG -p out/$NB"
----- run_phase4.sh (114 lines)
#!/usr/bin/env bash
# One command: pre-flight checks, then push the four Phase 4 notebooks
# (a_cpu, a_gpu, b_gpu, c_cpu).
#
#   ./run_phase4.sh --dry      build + check everything, push nothing
#   ./run_phase4.sh            push all four
#
# The one-shot seal notebook is NOT pushed here; run it last.

set -euo pipefail

export KUSER="${KUSER:-yoka33}"
DRY="${1:-}"

cd "$(dirname "$0")"
chmod +x push_phase4.sh

echo "== pre-flight =="

# Locate the Phase 4 source script
SRCF="af_phase4.py"
[ -f "$SRCF" ] || SRCF="code/af_phase4.py"
[ -f "$SRCF" ] || {
  echo "MISSING af_phase4.py (put it in the repo root or code/)"
  exit 1
}

[ -f "push_phase4.sh" ] || {
  echo "MISSING push_phase4.sh"
  exit 1
}

if [ "$DRY" != "--dry" ]; then
  command -v kaggle >/dev/null || {
    echo "kaggle CLI not found: pip install kaggle"
    exit 1
  }

  # Do not fail merely because ~/.kaggle/kaggle.json is absent.
  # The Kaggle CLI may use another config location or environment variables.
  # Test authentication with a lightweight API call instead.
  if kaggle kernels list --mine --page-size 1 >/dev/null 2>&1; then
    echo "kaggle authentication OK"
  else
    echo "Kaggle authentication failed."
    echo "Check one of these:"
    echo "  1. kaggle.json in ~/.kaggle/"
    echo "  2. KAGGLE_USERNAME and KAGGLE_KEY environment variables"
    echo "  3. KAGGLE_CONFIG_DIR pointing to a directory containing kaggle.json"
    echo "  4. Run: kaggle configure"
    exit 1
  fi

  # Competition access is useful but should not block a kernel push
  # if the competition rules have already been accepted in the browser.
  if kaggle competitions files talkingdata-adtracking-fraud-detection >/dev/null 2>&1; then
    echo "competition access OK"
  else
    echo "WARNING: could not list competition files."
    echo "If kernel pushes work, this is usually harmless."
    echo "If needed, accept the rules at:"
    echo "  https://www.kaggle.com/c/talkingdata-adtracking-fraud-detection/rules"
  fi

  echo "kaggle OK; user $KUSER"
fi

echo
echo "== local sanity (grammar, about 10 seconds) =="

python3 "$SRCF" --dry-run --n 60 2>&1 |
  grep -oE "dry-run rung=[0-9]: grammar_ok=[A-Za-z]+|effective_atomic': [0-9]+|total_effective': [0-9]+" |
  paste -sd' ' |
  sed 's/dry-run/\n dry-run/g'

echo
echo "== building and pushing notebooks =="

for nb in a_cpu a_gpu b_gpu c_cpu; do
  echo
  echo "== $nb =="
  ./push_phase4.sh "$nb" $DRY
done

cat <<EOF

== next ==

status:
  for n in a-cpu a-gpu b-gpu c-cpu; do
    kaggle kernels status $KUSER/af-phase4-\$n
  done

outputs:
  kaggle kernels output $KUSER/af-phase4-a-cpu -p out_phase4/af-phase4-a-cpu
  kaggle kernels output $KUSER/af-phase4-a-gpu -p out_phase4/af-phase4-a-gpu
  kaggle kernels output $KUSER/af-phase4-b-gpu -p out_phase4/af-phase4-b-gpu
  kaggle kernels output $KUSER/af-phase4-c-cpu -p out_phase4/af-phase4-c-cpu

resume:
  If a notebook stops at the time guard, attach its own previous output
  and re-push it:

  KSRC=$KUSER/af-phase4-a-cpu ./push_phase4.sh a_cpu
  KSRC=$KUSER/af-phase4-a-gpu ./push_phase4.sh a_gpu
  KSRC=$KUSER/af-phase4-b-gpu ./push_phase4.sh b_gpu
  KSRC=$KUSER/af-phase4-c-cpu ./push_phase4.sh c_cpu

LAST, once all four are done and the plan is committed:

  KSRC=$KUSER/af-phase4-a-cpu,$KUSER/af-phase4-a-gpu,$KUSER/af-phase4-b-gpu,$KUSER/af-phase4-c-cpu \
  P3DATA=$KUSER/af-phase3-results \
  ./push_phase4.sh seal
EOF
